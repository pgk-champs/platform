# План: Зал показывает все 46 механик

> **Для исполнителя:** задачи идут по порядку, шаги отмечаются `- [ ]`. Каждая
> задача заканчивается рабочим коммитом — остановиться можно после любой.

**Цель:** зал показывает все 46 механик платформы вместо одиннадцати, а список
тренажёров генерится из глав, а не ведётся руками.

**Как:** `knowledge-map.mjs` пишет `src/data/trainers.json` по блокам
`<Block kind="trainer">`; подписи механик лежат отдельным файлом под стражем;
страница зала строится из этих двух источников. Вкладка сворачивается в три
пункта, тренажёры входят в счёт наполнения главы.

**Стек:** Docusaurus 3.10.2, TypeScript, `node --test` для скриптов, vitest
для компонентов.

Спека: `specs/2026-09-17-zal.md`.

## Общие ограничения

- **Коммит не упоминает Claude, AI и соавторство.** Сообщение по-русски, от
  лица Олега: что изменилось и почему.
- **Перед коммитом грепнуть диф:**
  `git diff HEAD | grep -nEi "(-----BEGIN|api[_-]?key|secret|passw|ssh-rsa|AKIA|xox[baprs]-)"`
- **Пять гейтов до пуша:** `npm run kmap`, `npm test`, `npm run typecheck`,
  `npx vitest run`, `npm run build`. Пуш в `main` = публикация на прод.
- **Цвета по ролям, менять нельзя:** медь `--pgk-you` — пройденное и твой
  прогресс; индиго `--ifm-color-primary` — структура (ссылка, активный пункт,
  граница); `--pgk-track-*` — только чип и тонкая полоса, заливкой нет;
  зелёный и красный — только «сошлось / не сошлось».
- **`--pgk-you-soft` в `custom.css` НЕ существует**, хотя спека переоформления
  его упоминала. Подложка меди делается через `color-mix(in srgb,
  var(--pgk-you) 18%, transparent)`, новый токен не заводится.
- **Радиус:** карточка `var(--ifm-card-border-radius)` (14px), мелочь
  `var(--pgk-radius-sm)` (8px).
- **vitest не резолвит `@theme/*` и `@docusaurus/*`** — вся чистая логика
  живёт в `src/lib/`, там же тестируется.
- **cwd сбрасывается между вызовами Bash** — каждая команда начинается с
  `cd …/platform &&`.
- **Дев-сервер только через Browser pane** (`preview_start`), не через Bash.

```js
// Реестр тренажёров для Зала. Собирается по блокам <Block kind="trainer">:
// проверено по всем главам — ВСЕ trainerId лежат внутри такого блока, других
// мест нет. Здесь только то, что расходится при каждой правке глав; подпись
// механики генерировать нельзя (резать title по двоеточию даёт трём разным
// механикам имя «Квест»), она лежит руками в src/data/trainer-names.ts.
export function buildTrainers(docsDir) {
  const byComponent = new Map();
  const walk = (d) =>
    fs.readdirSync(d, { withFileTypes: true }).forEach((e) => {
      const p = path.join(d, e.name);
      if (e.isDirectory()) return walk(p);
      if (!/\.mdx?$/.test(e.name) || /^index\.mdx?$/.test(e.name)) return;
      const { data, content } = matter(fs.readFileSync(p, 'utf8'));
      if ((data.kind || 'chapter') !== 'chapter') return;

      const base = e.name.replace(/\.mdx?$/, '');
      const fileId = stripNumberPrefix(base);
      const rel = path.relative(docsDir, d).split(path.sep).filter(Boolean);
      const track = rel[0] || '';
      const docPath = [...rel.map(stripNumberPrefix), fileId].join('/');

      const blocks = /<Block\s+([^>]*)>([\s\S]*?)<\/Block>/g;
      let b;
      while ((b = blocks.exec(content))) {
        if (!/kind="trainer"/.test(b[1])) continue;
        // Блок-тренажёр без trainerId прогресс не пишет и в зал не идёт —
        // так, например, в linux-terminal, где блоков семь, а тренажёров шесть.
        const inner = /<([A-Z]\w*)((?:[^>"]|"[^"]*")*?)trainerId="([^"]+)"/.exec(b[2]);
        if (!inner) continue;
        const [, component, props, trainerId] = inner;
        const list = byComponent.get(component) ?? [];
        list.push({
          title: /title="([^"]*)"/.exec(b[1])?.[1] ?? '',
          // chapterId берём с самого тренажёра: он не всегда равен имени файла
          // (02b-english-practice → english-practice), а прогресс пишется по нему.
          chapterId: /chapterId="([^"]+)"/.exec(props)?.[1] ?? fileId,
          track,
          path: docPath,
          blockId: /blockId="([^"]+)"/.exec(b[1])?.[1] ?? trainerId,
          trainerId,
        });
        byComponent.set(component, list);
      }
    });
  walk(docsDir);
  return [...byComponent]
    .map(([component, exercises]) => ({ component, exercises }))
    .sort((a, b) => b.exercises.length - a.exercises.length || a.component.localeCompare(b.component));
}
```

import trainers from '../data/trainers.json';
import knowledgeMap from '../data/knowledge-map.json';
import { TRAINER_NAMES } from '../data/trainer-names';
import { store } from './store';

// Логика Зала отдельно от вёрстки: vitest не резолвит @docusaurus/*, а проверять
// сборку карточек, поиск и чипы надо. Компонент отсюда только читает.

type Exercise = {
  title: string; chapterId: string; track: string;
  path: string; blockId: string; trainerId: string;
};
type Mechanic = { component: string; exercises: Exercise[] };

export type GymCard = {
  component: string;
  name: string;
  blurb: string;
  tracks: string[];
  count: number;
  runnable: boolean;
  exercises: { title: string; href: string; chapterTitle: string }[];
  done: number;
};

// Механики, у которых есть демо-данные и которые запускаются прямо в зале.
// Остальные 35 честно ведут в главу: половина из них осмысленна только там.
export const RUNNABLE = [
  'CodeTyping', 'TerminalSim', 'GitSim', 'HashPlayground', 'BlockChainDemo',
  'SignDemo', 'ComposePreview', 'ChmodCalc', 'HotkeyTrainer', 'WordOrder',
  'PredictOutput',
];

const TITLE_BY_ID: Record<string, string> = Object.fromEntries(
  (knowledgeMap as { id: string; title: string }[]).map((e) => [e.id, e.title]),
);

export function buildCards(): GymCard[] {
  const progress = store.getProgress();
  return (trainers as Mechanic[]).map((m) => {
    const meta = TRAINER_NAMES[m.component] ?? { name: m.component, blurb: '' };
    return {
      component: m.component,
      name: meta.name,
      blurb: meta.blurb,
      tracks: [...new Set(m.exercises.map((e) => e.track))],
      count: m.exercises.length,
      runnable: RUNNABLE.includes(m.component),
      exercises: m.exercises.map((e) => ({
        // У 85 упражнений из 210 своего title нет — почти все это наборы
        // слепой печати. Там подписываемся главой, а не пустой строкой.
        title: e.title || TITLE_BY_ID[e.chapterId] || e.chapterId,
        href: `/docs/${e.path}#${e.blockId}`,
        chapterTitle: TITLE_BY_ID[e.chapterId] ?? e.chapterId,
      })),
      done: m.exercises.filter(
        (e) => progress.trainers[e.chapterId]?.[e.trainerId],
      ).length,
    };
  });
}

export function filterCards(cards: GymCard[], q: string, chip: string): GymCard[] {
  const needle = q.trim().toLowerCase();
  return cards.filter((c) => {
    if (chip === 'runnable' && !c.runnable) return false;
    if (chip !== 'all' && chip !== 'runnable' && !c.tracks.includes(chip)) return false;
    if (!needle) return true;
    return `${c.name} ${c.blurb}`.toLowerCase().includes(needle);
  });
}

export function chipCounts(cards: GymCard[]): { id: string; label: string; n: number }[] {
  const TRACK_LABEL: Record<string, string> = {
    foundation: 'Фундамент', mobile: 'Мобилка',
    blockchain: 'Блокчейн', advanced: 'Отдельные темы',
  };
  return [
    { id: 'all', label: 'Всё', n: cards.length },
    ...Object.entries(TRACK_LABEL).map(([id, label]) => ({
      id, label, n: cards.filter((c) => c.tracks.includes(id)).length,
    })),
    { id: 'runnable', label: 'Запускаются здесь', n: cards.filter((c) => c.runnable).length },
  ];
}

// Наполнение главы считается по делам, а не по отметкам: ручной галочки
// «пройдено» на платформе больше нет. Тренажёры входят в счёт с 17.09.2026 —
// до этого в шапке главы висело «Тренажёры 0/6», а сосуд наливался до крышки,
// и одно слово значило две разные вещи.

```ts
export function fillOf(chapterId: string): number {
  const t = TOTALS[chapterId];
  if (!t) return 0;
  const denom = t.sections + t.quizzes + t.trainers;
  if (denom === 0) return 0;

  const progress = store.getProgress();
  const sections = progress.sections[chapterId]?.length ?? 0;
  // Проверка засчитывается только целиком верной — так же, как это понимало
  // прежнее «пройдена». Иначе сосуд наполнялся бы провалами.
  const quizzes = Object.values(progress.quizzes[chapterId] ?? {}).filter(
    (q) => q.correct === q.total,
  ).length;
  // Тренажёр засчитывается фактом прохождения: у разных механик нет общей
  // шкалы «верно», и требовать её значило бы переписывать все сорок шесть.
  const trainers = Math.min(
    Object.keys(progress.trainers[chapterId] ?? {}).length,
    t.trainers,
  );

  return Math.min(1, (sections + quizzes + trainers) / denom);
}
```
